// control de giro proporcional simple con Limelight.
// el "control proporcional" es un algoritmo de control en el que la salida es proporcional al error.
// en este caso, vamos a devolver una velocidad angular que es proporcional al
// valor "tx" de Limelight.
double
limelight_aim_proportional
(
)
{
// kP (constante de proporcionalidad)
// este es un número ajustado manualmente que determina la agresividad de nuestro bucle de control proporcional
// si es demasiado alto, el robot oscilará.
// si es demasiado bajo, el robot nunca alcanzará su objetivo
// si el robot nunca gira en la dirección correcta, kP debe invertirse.
double
kP
=
.035
;
// tx varía de (-hfov/2) a (hfov/2) en grados. Si tu objetivo está en el borde más a la derecha
// de tu feed de limelight 3, tx debería devolver aproximadamente 31 grados.
double
targetingAngularVelocity
=
LimelightHelpers
.
getTX
(
"limelight"
)
*
kP
;
// convertir a radianes por segundo para nuestro método de conducción
targetingAngularVelocity
*=
Drivetrain
.
kMaxAngularSpeed
;
// invertir ya que tx es positivo cuando el objetivo está a la derecha de la cruz
targetingAngularVelocity
*=
-
1.0
;
return
targetingAngularVelocity
;
}
// control proporcional simple de distancia con el valor "ty" de Limelight
// esto funciona mejor si la altura de montaje de tu Limelight y la altura de montaje del objetivo son diferentes.
// si tu limelight y el objetivo están montados a la misma o similar altura, usa "ta" (área) para el ajuste de distancia en lugar de "ty"

1.0
;

targetingForwardSpeed
;
